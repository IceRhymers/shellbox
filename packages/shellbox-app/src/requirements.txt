# --- what the relay needs ---

# `websockets` is NOT preinstalled on the Apps runtime, and uvicorn needs it to serve the
# WebSocket handshake at all. Without it the upgrade fails inside the App, and a caller reads
# that as the edge blocking WebSockets. The Phase 1 probe avoided that misdiagnosis by
# declaring it, and so does this.
websockets==15.0.1

# fastapi and uvicorn ARE preinstalled on the Apps runtime, and they are declared anyway.
# `shellbox_app.server` imports fastapi at module scope, so the import check cannot run
# without them -- and a check that borrowed the runtime's copy would be measuring a different
# environment from the one this file describes. Declaring them means pip installs these pins
# over whatever the runtime preinstalls, which is what ADR-22 asks for.
fastapi==0.140.7
uvicorn==0.51.0

# --- what the registry needs ---
#
# `shellbox_registry` joined the deploy root here. `shellbox_app.database` imports it at module
# scope, so these are not optional extras: without them the App boots into
# `ModuleNotFoundError` behind a green deploy. The list is `shellbox-registry`'s own
# `[project] dependencies` in ../../shellbox-registry/pyproject.toml, plus its `lakebase`
# extra, pinned.

# SQLAlchemy is the registry's ORM, and `shellbox_registry.postgres` imports it at module
# scope.
sqlalchemy==2.0.51

# The Postgres driver, and the `[binary]` extra is load-bearing rather than a preference: it
# ships a prebuilt wheel. Plain `psycopg` builds against libpq at install time, and the Apps
# runtime is not a build environment.
psycopg[binary]==3.3.4

# alembic is a `shellbox-registry` dependency because the package carries the migrations. The
# App never migrates -- `scripts/deploy.sh` step 3 does, as the deploying principal -- but the
# staged root holds the whole package, `shellbox_registry/alembic/` included, and a partial
# copy of a package is a worse thing to deploy than an unused import.
alembic==1.18.5

# The `lakebase` extra of `shellbox-registry`, written out because pip reads a flat list here.
# `lakebase.py` mints an OAuth token to use as the Postgres password, and that needs the SDK.
#
# WARNING: no import check can verify this line. `lakebase.py` imports the SDK LAZILY by
# design, so `import shellbox_registry.lakebase` succeeds on a runtime with no SDK at all.
# `scripts/deploy-app.sh` says the same thing where the check is written. The first thing that
# actually needs the SDK is minting a token against the live endpoint.
databricks-sdk==0.123.0
